===== src\contexts\RaffleContext.jsx =====
import {
  createContext,
  useCallback,
  useMemo,
  useReducer,
  useRef,
} from 'react'

const uid = () => Math.random().toString(36).slice(2) + Date.now()

const initialState = {
  participants: [],
  winners: [],
  winnersCount: 1,
  status: 'idle',
  draws: 0,
}

function reducer(state, action) {
  switch (action.type) {
    case 'ADD':
      return { ...state, participants: [...state.participants, action.payload], winners: [] }
    case 'ADD_MANY':
      return { ...state, participants: [...state.participants, ...action.payload], winners: [] }
    case 'REMOVE':
      return {
        ...state,
        participants: state.participants.filter((p) => p.id !== action.payload),
        winners: state.winners.filter((w) => w.id !== action.payload),
      }
    case 'CLEAR':
      return { ...initialState, winnersCount: state.winnersCount, draws: state.draws }
    case 'SET_COUNT':
      return { ...state, winnersCount: Math.max(1, Number(action.payload) || 1) }
    case 'START':
      return { ...state, status: 'spinning', winners: [] }
    case 'SET_WINNERS':
      return {
        ...state,
        status: 'done',
        winners: action.payload,
        draws: state.draws + 1,
      }
    case 'RESET':
      return { ...state, status: 'idle', winners: [] }
    default:
      return state
  }
}

export const RaffleContext = createContext(null)

export function RaffleProvider({ children }) {
  const [state, dispatch] = useReducer(reducer, initialState)
  const stateRef = useRef(state)
  stateRef.current = state

  const addParticipant = useCallback((raw) => {
    const value = String(raw ?? '').trim()
    if (!value) return false
    const current = stateRef.current.participants
    const isNum = /^\d+$/.test(value)

    if (isNum && current.some((p) => p.number === Number(value))) return false
    if (!isNum && current.some((p) => p.name.toLowerCase() === value.toLowerCase())) return false

    const number = isNum
      ? Number(value)
      : current.reduce((m, p) => Math.max(m, p.number), 0) + 1

    dispatch({
      type: 'ADD',
      payload: { id: uid(), name: isNum ? `Pelota ${number}` : value, number },
    })
    return true
  }, [])

  const loadMany = useCallback((list) => {
    const current = stateRef.current.participants
    const base = current.length
    const nuevos = list.map((raw, i) => {
      const value = String(raw ?? '').trim() || `Invitado ${base + i + 1}`
      const isNum = /^\d+$/.test(value)
      const number = isNum
        ? Number(value)
        : current.reduce((m, p) => Math.max(m, p.number), 0) + i + 1
      return { id: uid(), name: isNum ? `Pelota ${number}` : value, number }
    })
    dispatch({ type: 'ADD_MANY', payload: nuevos })
  }, [])

  const removeParticipant = useCallback((id) => dispatch({ type: 'REMOVE', payload: id }), [])
  const clearAll = useCallback(() => dispatch({ type: 'CLEAR' }), [])
  const setWinnersCount = useCallback((n) => dispatch({ type: 'SET_COUNT', payload: n }), [])

  const drawWinners = useCallback(() => {
    const snap = stateRef.current
    if (snap.status === 'spinning') return
    if (snap.participants.length === 0) return

    dispatch({ type: 'START' })

    const pool = [...snap.participants]
    const count = Math.min(snap.winnersCount, pool.length)
    const picked = []
    for (let i = 0; i < count; i += 1) {
      const idx = Math.floor(Math.random() * pool.length)
      picked.push(pool.splice(idx, 1)[0])
    }

    setTimeout(() => {
      dispatch({ type: 'SET_WINNERS', payload: picked })
    }, 2600)
  }, [])

  const resetDraw = useCallback(() => dispatch({ type: 'RESET' }), [])

  const value = useMemo(
    () => ({
      participants: state.participants,
      winners: state.winners,
      winnersCount: state.winnersCount,
      status: state.status,
      draws: state.draws,
      addParticipant,
      loadMany,
      removeParticipant,
      clearAll,
      setWinnersCount,
      drawWinners,
      resetDraw,
      dispatch,
    }),
    [state, addParticipant, loadMany, removeParticipant, clearAll, setWinnersCount, drawWinners, resetDraw]
  )

  return (
    <RaffleContext.Provider value={value}>
      {children}
    </RaffleContext.Provider>
  )
}

import { createContext, useCallback, useEffect, useMemo, useState } from 'react'

export const ThemeContext = createContext(null)

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(() => {
    if (typeof window === 'undefined') return 'light'
    return (
      localStorage.getItem('theme') ||
      (window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light')
    )
  })

  useEffect(() => {
    document.documentElement.setAttribute('data-theme', theme)
    localStorage.setItem('theme', theme)
  }, [theme])

  const toggleTheme = useCallback(() => {
    setTheme((t) => (t === 'light' ? 'dark' : 'light'))
  }, [])

  const value = useMemo(() => ({ theme, setTheme, toggleTheme }), [theme, toggleTheme])

  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>
}


===== src\contexts\ToastContext.jsx =====
import { createContext, useCallback, useEffect, useMemo, useRef, useState } from 'react'

export const ToastContext = createContext(null)

let seq = 0

export function ToastProvider({ children }) {
  const [toasts, setToasts] = useState([])
  const timers = useRef(new Map())

  const remove = useCallback((id) => {
    setToasts((t) => t.filter((x) => x.id !== id))
    const timer = timers.current.get(id)
    if (timer) {
      clearTimeout(timer)
      timers.current.delete(id)
    }
  }, [])

  const notify = useCallback((message, type = 'info', ttl = 3200) => {
    const id = ++seq
    setToasts((t) => [...t.slice(-4), { id, message, type }])
    timers.current.set(id, setTimeout(() => remove(id), ttl))
    return id
  }, [remove])

  useEffect(() => {
    const map = timers.current
    return () => {
      map.forEach(clearTimeout)
      map.clear()
    }
  }, [])

  const value = useMemo(
    () => ({ toasts, notify, remove, addToast: notify, removeToast: remove }),
    [toasts, notify, remove]
  )

  return <ToastContext.Provider value={value}>{children}</ToastContext.Provider>
}

===== src\hooks\useForm.js =====
import { useCallback, useRef, useState } from 'react'

export function useForm(initialValues = {}) {
  // useRef guarda el valor inicial UNA sola vez → reset() es estable
  // aunque el padre pase un objeto nuevo en cada render.
  const initialRef = useRef(initialValues)

  const [values, setValues] = useState(initialValues)
  const [errors, setErrors] = useState({})

  const handleChange = useCallback((event) => {
    const { name, value } = event.target
    setValues((prev) => ({ ...prev, [name]: value }))
    setErrors((prev) => (prev[name] ? { ...prev, [name]: undefined } : prev))
  }, [])

  const reset = useCallback(() => {
    setValues(initialRef.current)
    setErrors({})
  }, [])

  const setFieldError = useCallback((field, message) => {
    setErrors((prev) => ({ ...prev, [field]: message }))
  }, [])

  return { values, errors, handleChange, reset, setErrors, setFieldError }
}

export const BALL_COLORS = [
  ['#D946EF', '#8B1FBF'],  // fucsia
  ['#A855F7', '#6D28D9'],  // violeta
  ['#22D3EE', '#0891B2'],  // cian
  ['#34D399', '#059669'],  // verde
  ['#FBBF24', '#D97706'],  // dorado
  ['#F472B6', '#BE185D'],  // rosa
  ['#60A5FA', '#2563EB'],  // azul
  ['#FB7185', '#BE123C'],  // rojo
  ['#A78BFA', '#7C3AED'],  // lavanda
  ['#F97316', '#C2410C'],  // naranja
]

export function colorForNumber(number) {
  const idx = Math.abs(Number(number) || 0) % BALL_COLORS.length
  return BALL_COLORS[idx]
}

export default function BallForm() {
  const { addParticipant, loadMany, participants } = useRaffle()
  const { values, errors, handleChange, reset, setFieldError } = useForm({ entry: '' })
  const inputRef = useRef(null)

  useEffect(() => { inputRef.current?.focus() }, [])

  const handleSubmit = (event) => {
    event.preventDefault()

    const base = participants.length

    <section className="card">
      <header className="card__head">
        <h2 className="card__title">1 · Agrega pelotas al bombo</h2>
        <RenderBadge label="BallForm" />
      </header>

      <form className="row" onSubmit={handleSubmit}>
        <input
          ref={inputRef}
          name="entry"
          value={values.entry}
          onChange={handleChange}
          placeholder="N.º 7  ó  Ana"
          autoComplete="off"
          className={errors.entry ? 'input input--error' : 'input'}
        />
        <button className="btn btn--primary" type="submit">
          <IconPlus /> Agregar
        </button>
        <button className="btn" type="button" onClick={quickAdd}>
          <IconBolt /> +5 rápido
        </button>

        Si escribes un <b>nombre</b>, se le asigna el siguiente número libre.
      </p>
    </section>
  )
}

export default function BallList() {
  const { participants, winners, removeParticipant } = useRaffle()
  const [query, setQuery] = useState('')

  const winnerIds = useMemo(() => new Set(winners.map((w) => w.id)), [winners])

          onChange={(e) => setQuery(e.target.value)}
        />
      </div>

      {participants.length === 0 ? (
        <p className="empty">El bombo está vacío. Agrega tu primera pelota.</p>

        <p className="empty">Sin resultados para «{query}».</p>

import { useMemo, useState } from 'react'
import { useRaffle } from '../contexts/RaffleContext'
import RenderBadge from './RenderBadge'

const ITERACIONES = 4_000_000

const IconChart = () => (
  <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"
       strokeLinecap="round" strokeLinejoin="round" width="16" height="16" aria-hidden="true">
    <path d="M3 3v18h18" />
    <path d="M7 15V9M12 15V5M17 15v-7" />
  </svg>
)

export default function StatsPanel() {
  const { participants, winnersCount } = useRaffle()
  const [heavy, setHeavy] = useState(false)
  const [broken, setBroken] = useState(false)

  // useMemo #1 — estadísticas baratas
  const stats = useMemo(() => {
    const total = participants.length
    const prob = total > 0 ? Math.min(1, winnersCount / total) : 0
    const sum = participants.reduce((acc, p) => acc + p.number, 0)
    return { total, prob, sum, avg: total ? sum / total : 0 }
  }, [participants, winnersCount])

  // useMemo #2 — cálculo deliberadamente pesado
  const heavyResult = useMemo(() => {
    if (!heavy) return null
    const t0 = performance.now()
    let acc = 0
    for (let i = 0; i < ITERACIONES; i += 1) acc += Math.sqrt(i)
    return { ms: performance.now() - t0, sample: acc.toFixed(2) }
  }, [heavy, participants.length])

  // Solo se lanza si TÚ presionas el botón
  if (broken) throw new Error('Error de prueba lanzado desde StatsPanel')

  return (
    <section className="card card--stats">
      <header className="card__head">
        <h2 className="card__title">
          <IconChart /> Métricas del bombo
        </h2>
        <RenderBadge label="Stats" />
      </header>

      <div className="grid-4">
        <div className="metric"><span>Pelotas</span><b>{stats.total}</b></div>
        <div className="metric"><span>Premios</span><b>{winnersCount}</b></div>
        <div className="metric"><span>Prob. ganar</span><b>{(stats.prob * 100).toFixed(1)}%</b></div>
        <div className="metric"><span>N.º promedio</span><b>{stats.avg.toFixed(1)}</b></div>
      </div>

      <label className="check">
        <input
          type="checkbox"
          checked={heavy}
          onChange={(e) => setHeavy(e.target.checked)}
        />
        Activar cálculo pesado ({ITERACIONES.toLocaleString('es-PE')} iteraciones)
      </label>

      {heavyResult && (
        <p className="hint">
          Último cálculo: <b>{heavyResult.ms.toFixed(1)} ms</b> · muestra {heavyResult.sample}
          <br />
          Agrega una pelota y verás que se recalcula. Cambia el tema o escribe en el filtro y
          verás que <b>NO</b> se recalcula.
        </p>
      )}

      <button
        className="btn btn--ghost"
        type="button"
        onClick={() => setBroken(true)}
      >
        Probar ErrorBoundary
      </button>
    </section>
  )
}

import { Component } from 'react'

export default class ErrorBoundary extends Component {
  state = { error: null }

  static getDerivedStateFromError(error) {
    return { error }
  }

  componentDidCatch(error, info) {
    this.info = info
    // Aquí podrías enviar el error a un servicio de monitoreo
    console.error('[ErrorBoundary]', error, info)
  }

  handleReset = () => {
    this.setState({ error: null })
    // Recarga la página para limpiar el estado que causó el error
    window.location.reload()
  }

  render() {
    if (this.state.error) {
      return (
        <div className="crash">
          <div className="crash__icon" aria-hidden="true">
            <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"
                 strokeLinecap="round" strokeLinejoin="round" width="32" height="32">
              <path d="M10.3 3.9 1.8 18a2 2 0 0 0 1.7 3h17a2 2 0 0 0 1.7-3L13.7 3.9a2 2 0 0 0-3.4 0Z" />
              <path d="M12 9v4" />
              <path d="M12 17h.01" />
            </svg>
          </div>
          <h1 className="crash__title">Algo se rompió en el sorteo</h1>
          <pre className="crash__msg">
            {String(this.state.error?.message ?? this.state.error)}
          </pre>
          <button className="btn btn--primary" onClick={this.handleReset} type="button">
            Reintentar
          </button>
        </div>
      )
    }
    return this.props.children
  }
}
